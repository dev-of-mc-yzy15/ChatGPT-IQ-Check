    <button class="toggle" id="motionToggle" type="button" aria-pressed="false" aria-controls="artwork" aria-label="暂停动画">
      <i class="toggle-icon" aria-hidden="true"></i>
      <span id="toggleText">暂停动画</span>
    </button>
  </main>

  <script>
    (() => {
      const scene = document.getElementById('scene');
      const toggle = document.getElementById('motionToggle');
      const toggleText = document.getElementById('toggleText');
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');

      const nodes = {
        body: document.getElementById('birdBody'),
        rearSpokes: document.getElementById('rearSpokes'),
        frontSpokes: document.getElementById('frontSpokes'),
        rearOutline: document.getElementById('rearLegOutline'),
        rearLeg: document.getElementById('rearLeg'),
        frontOutline: document.getElementById('frontLegOutline'),
        frontLeg: document.getElementById('frontLeg'),
        rearFoot: document.getElementById('rearFoot'),
        frontFoot: document.getElementById('frontFoot'),
        crankArm: document.getElementById('crankArm'),
        rearPedal: document.getElementById('rearPedal'),
        frontPedal: document.getElementById('frontPedal')
      };

      const crank = { x: 575, y: 550, radius: 39 };
      const rearHip = { x: 528, y: 395 };
      const frontHip = { x: 555, y: 397 };
      let playing = !reducedMotion.matches;
      let userOverride = false;
      let phase = -0.65;
      let lastTime = performance.now();
      let frameId = 0;

      // A compact two-link inverse-kinematics solver keeps each foot on its pedal.
      function solveLeg(hip, foot, upper, lower, bend) {
        const dx = foot.x - hip.x;
        const dy = foot.y - hip.y;
        const rawDistance = Math.hypot(dx, dy) || 1;
        const distance = Math.min(upper + lower - 0.1, Math.max(Math.abs(upper - lower) + 0.1, rawDistance));
        const ux = dx / rawDistance;
        const uy = dy / rawDistance;
        const along = (upper * upper - lower * lower + distance * distance) / (2 * distance);
        const height = Math.sqrt(Math.max(0, upper * upper - along * along));
        return {
          x: hip.x + along * ux + bend * height * -uy,
          y: hip.y + along * uy + bend * height * ux
        };
      }

      function pointOnCrank(angle) {
        return {
          x: crank.x + Math.cos(angle) * crank.radius,
          y: crank.y + Math.sin(angle) * crank.radius
        };
      }

      function pathFor(hip, knee, foot) {
        return `M${hip.x.toFixed(2)} ${hip.y.toFixed(2)} L${knee.x.toFixed(2)} ${knee.y.toFixed(2)} L${foot.x.toFixed(2)} ${foot.y.toFixed(2)}`;
      }

      function setLeg(path, outline, hip, foot, bend) {
        const knee = solveLeg(hip, foot, 100, 106, bend);
        const data = pathFor(hip, knee, foot);
        path.setAttribute('d', data);
        outline.setAttribute('d', data);
      }

      function setPedal(line, point) {
        line.setAttribute('x1', point.x - 20);
        line.setAttribute('x2', point.x + 20);
        line.setAttribute('y1', point.y);
        line.setAttribute('y2', point.y);
      }

      function draw() {
        const bob = Math.sin(phase * 2) * 2.6;
        const tilt = Math.sin(phase * 2) * 0.28;
        nodes.body.setAttribute('transform', `translate(0 ${bob.toFixed(2)}) rotate(${tilt.toFixed(2)} 560 370)`);

        const rearPoint = pointOnCrank(phase + Math.PI);
        const frontPoint = pointOnCrank(phase);
        const rearHipNow = { x: rearHip.x, y: rearHip.y + bob };
        const frontHipNow = { x: frontHip.x, y: frontHip.y + bob };

        setLeg(nodes.rearLeg, nodes.rearOutline, rearHipNow, rearPoint, 1);
        setLeg(nodes.frontLeg, nodes.frontOutline, frontHipNow, frontPoint, -1);
        nodes.rearFoot.setAttribute('transform', `translate(${rearPoint.x.toFixed(2)} ${rearPoint.y.toFixed(2)}) rotate(${(Math.sin(phase + Math.PI) * 8).toFixed(2)})`);
        nodes.frontFoot.setAttribute('transform', `translate(${frontPoint.x.toFixed(2)} ${frontPoint.y.toFixed(2)}) rotate(${(Math.sin(phase) * 8).toFixed(2)})`);

        nodes.crankArm.setAttribute('x1', rearPoint.x);
        nodes.crankArm.setAttribute('y1', rearPoint.y);
        nodes.crankArm.setAttribute('x2', frontPoint.x);
        nodes.crankArm.setAttribute('y2', frontPoint.y);
        setPedal(nodes.rearPedal, rearPoint);
        setPedal(nodes.frontPedal, frontPoint);

        const wheelDegrees = phase * 2.58 * 180 / Math.PI;
        nodes.rearSpokes.setAttribute('transform', `rotate(${wheelDegrees.toFixed(2)} 360 550)`);
        nodes.frontSpokes.setAttribute('transform', `rotate(${wheelDegrees.toFixed(2)} 860 550)`);
      }

      function frame(now) {
        const delta = Math.min(40, now - lastTime);
        lastTime = now;
        phase += delta * 0.00445;
        draw();
        frameId = playing ? requestAnimationFrame(frame) : 0;
      }

      function setPlaying(next, fromUser = false) {
        if (fromUser) userOverride = true;
        playing = next;
        scene.classList.toggle('paused', !playing);
        toggle.setAttribute('aria-pressed', String(!playing));
        toggleText.textContent = playing ? '暂停动画' : '播放动画';
        toggle.setAttribute('aria-label', playing ? '暂停动画' : '播放动画');

        if (playing && !frameId) {
          lastTime = performance.now();
          frameId = requestAnimationFrame(frame);
        } else if (!playing && frameId) {
          cancelAnimationFrame(frameId);
          frameId = 0;
          draw();
        }
      }

      toggle.addEventListener('click', () => setPlaying(!playing, true));
      const onMotionPreferenceChange = event => {
        if (!userOverride) setPlaying(!event.matches);
      };
      if (reducedMotion.addEventListener) {
        reducedMotion.addEventListener('change', onMotionPreferenceChange);
      } else {
        reducedMotion.addListener(onMotionPreferenceChange);
      }

      draw();
      setPlaying(playing);
    })();
  </script>
