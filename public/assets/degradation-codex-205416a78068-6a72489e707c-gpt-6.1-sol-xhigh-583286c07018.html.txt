  <script>
    (() => {
      'use strict';
      const $ = id => document.getElementById(id);
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      let running = !reducedMotion.matches;
      let speed = 1;
      let time = 0;
      let previousTime = null;
      let dusk = false;
      let animationFrame = null;
      const refs = Object.fromEntries(['rearWheel', 'frontWheel', 'chainring', 'pelican', 'farLeg', 'nearLeg', 'farFoot', 'nearFoot', 'farCrank', 'nearCrank', 'farPedal', 'nearPedal', 'hills', 'roadsideMoving', 'foregroundMoving', 'clouds', 'roadDashes', 'scarfTail', 'scarfFold', 'eye', 'bikeShadow', 'windLines', 'skyBirds'].map(id => [id, $(id)]));
      const TAU = Math.PI * 2;
      const set = (id, attr, value) => refs[id].setAttribute(attr, value);
      const point = (x, y) => `${x.toFixed(2)} ${y.toFixed(2)}`;

      // Two-link inverse kinematics keeps each foot on its moving pedal.
      function leg(hipX, hipY, footX, footY, upper, lower) {
        const dx = footX - hipX, dy = footY - hipY;
        const distance = Math.max(.01, Math.hypot(dx, dy));
        const along = (upper * upper - lower * lower + distance * distance) / (2 * distance);
        const bend = Math.sqrt(Math.max(0, upper * upper - along * along));
        const kneeX = hipX + along * dx / distance + bend * dy / distance;
        const kneeY = hipY + along * dy / distance - bend * dx / distance;
        return `M${point(hipX, hipY)} L${point(kneeX, kneeY)} L${point(footX, footY)}`;
      }

      function draw(t) {
        const phase = t * TAU / 1.65;
        const bob = Math.sin(phase * 2) * 1.5;
        const crankX = 591, crankY = 465, crankRadius = 31;
        const pedalX = crankX + Math.cos(phase) * crankRadius;
        const pedalY = crankY + Math.sin(phase) * crankRadius;
        const backX = crankX - Math.cos(phase) * crankRadius;
        const backY = crankY - Math.sin(phase) * crankRadius;
        const wheelDegrees = t * 78 / 80 * 180 / Math.PI;
        set('rearWheel', 'transform', `rotate(${wheelDegrees % 360})`);
        set('frontWheel', 'transform', `rotate(${wheelDegrees % 360})`);
        set('chainring', 'transform', `rotate(${phase * 180 / Math.PI % 360} 591 465)`);
        set('pelican', 'transform', `translate(0 ${bob})`);
        set('nearLeg', 'd', leg(575, 332 + bob, pedalX, pedalY, 88, 96));
        set('farLeg', 'd', leg(565, 331 + bob, backX, backY, 88, 96));
        set('nearCrank', 'd', `M591 465 L${point(pedalX, pedalY)}`);
        set('farCrank', 'd', `M591 465 L${point(backX, backY)}`);
        set('nearFoot', 'd', `M${point(pedalX - 10, pedalY - 5)}h21q10 0 12 7h-33Z`);
        set('farFoot', 'd', `M${point(backX - 10, backY - 5)}h21q10 0 12 7h-33Z`);
        set('nearPedal', 'd', `M${point(pedalX - 9, pedalY + 4)}h24`);
        set('farPedal', 'd', `M${point(backX - 9, backY + 4)}h24`);
        set('clouds', 'transform', `translate(${-((t * 6) % 1200)} 0)`);
        set('hills', 'transform', `translate(${-((t * 12) % 1200)} 0)`);
        set('roadsideMoving', 'transform', `translate(${-((t * 39) % 1200)} 0)`);
        set('foregroundMoving', 'transform', `translate(${-((t * 78) % 1200)} 0)`);
        set('roadDashes', 'stroke-dashoffset', (t * 78) % 101);
        const flap = Math.sin(t * 6) * 5;
        set('scarfTail', 'd', `M611 224C583 ${221 + flap} 562 ${202 - flap} 537 ${216 + flap}l8 10-10 8c33-9 48 ${15 - flap} 76 4Z`);
        set('scarfFold', 'd', `M543 ${222 + flap}q31 ${-6 - flap} 57 ${10 - flap}`);
        const blinkTime = t % 5.3;
        const blink = blinkTime > 4.95 && blinkTime < 5.12;
        set('eye', 'transform', blink ? 'translate(0 144) scale(1 .1)' : 'translate(0 0)');
        set('bikeShadow', 'rx', 201 + bob * 1.5);
        set('windLines', 'opacity', .45 + .2 * Math.sin(t * 3));
        set('windLines', 'transform', `translate(${-((t * 10) % 15)} 0)`);
        set('skyBirds', 'transform', `translate(${Math.sin(t * .3) * 12} ${Math.sin(t * .8) * 3})`);
      }

      function frame(timestamp) {
        animationFrame = null;
        if (!running || document.hidden) { previousTime = null; return; }
        if (previousTime !== null) time += Math.min((timestamp - previousTime) / 1000, .05) * speed;
        previousTime = timestamp;
        draw(time);
        animationFrame = requestAnimationFrame(frame);
      }

      function startLoop() {
        if (running && !document.hidden && animationFrame === null) {
          previousTime = null;
          animationFrame = requestAnimationFrame(frame);
        }
      }

      function updateControls(announce = false) {
        $('playIcon').setAttribute('d', running ? 'M4 2h3v12H4zM10 2h3v12h-3z' : 'M4 2 14 8 4 14Z');
        $('playText').textContent = running ? '歇一会儿' : '继续出发';
        $('playButton').setAttribute('aria-label', running ? '暂停骑行' : '继续骑行');
        $('playButton').setAttribute('aria-pressed', String(!running));
        $('rideStatus').textContent = running ? '微风徐徐 · 自在骑行中' : '停下来，也是一种风景';
        if (announce) $('announcement').textContent = running ? '骑行已继续' : '骑行已暂停';
      }

      function togglePlayback() {
        running = !running;
        if (!running && animationFrame !== null) { cancelAnimationFrame(animationFrame); animationFrame = null; }
        updateControls(true);
        startLoop();
      }

      $('playButton').addEventListener('click', togglePlayback);
      document.querySelectorAll('.speed-button').forEach(button => {
        button.addEventListener('click', () => {

          document.querySelectorAll('.speed-button').forEach(option => option.setAttribute('aria-pressed', String(option === button)));

        });
      });

        dusk = !dusk;
        $('scene').classList.toggle('dusk', dusk);

        $('themeText').textContent = dusk ? '回到晴朗时' : '等一场日落';

      });
      document.addEventListener('keydown', event => {
        if (event.code === 'Space' && !event.repeat && !event.altKey && !event.ctrlKey && !event.metaKey && !event.target.closest('button, input, textarea, select, a, [contenteditable]')) {
          event.preventDefault();
          togglePlayback();
        }
      });
      document.addEventListener('visibilitychange', () => {
        previousTime = null;
        if (document.hidden && animationFrame !== null) { cancelAnimationFrame(animationFrame); animationFrame = null; }
        else startLoop();
      });
      reducedMotion.addEventListener('change', event => {
        if (event.matches && running) togglePlayback();
      });
      draw(0);
      updateControls();
      startLoop();
    })();
