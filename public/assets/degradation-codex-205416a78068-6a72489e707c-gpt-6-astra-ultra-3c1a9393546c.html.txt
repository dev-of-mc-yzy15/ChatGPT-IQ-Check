  <script>
    (() => {
      'use strict';
      const $ = id => document.getElementById(id);
      const motionPreference = window.matchMedia('(prefers-reduced-motion: reduce)');
      const playButton = $('play-button');
      const speedButtons = [...document.querySelectorAll('[data-speed]')];
      let playing = !motionPreference.matches;
      let speed = 1;
      let elapsed = 0;
      let previousTime = null;
      let frameId = null;
      let bellStart = null;
      let audioContext;
      const parts = Object.fromEntries(['rear-wheel','front-wheel','crank','pelican','near-leg-path','near-leg-color','far-leg-path','near-foot','far-foot','clouds','birds','water-lines','road-movement','distant-plants','scarf-tail','eye','bell-rings'].map(id => [id, $(id)]));

      // Two-segment inverse kinematics keeps each foot attached to its pedal.
      function legPath(hipX, hipY, footX, footY) {
        const upper = 91, lower = 112;
        const dx = footX - hipX, dy = footY - hipY;
        const distance = Math.hypot(dx, dy);
        const along = (upper * upper - lower * lower + distance * distance) / (2 * distance);
        const height = Math.sqrt(Math.max(0, upper * upper - along * along));
        const kneeX = hipX + along * dx / distance + height * dy / distance;
        const kneeY = hipY + along * dy / distance - height * dx / distance;
        return `M${hipX.toFixed(2)} ${hipY.toFixed(2)}L${kneeX.toFixed(2)} ${kneeY.toFixed(2)}L${footX.toFixed(2)} ${footY.toFixed(2)}`;
      }

      function draw() {
        const phase = elapsed * 2.65;
        const degrees = phase * 180 / Math.PI;
        const bob = Math.sin(phase * 2) * 1.8;
        const crankX = Math.cos(phase) * 33;
        const crankY = Math.sin(phase) * 33;
        parts['rear-wheel'].setAttribute('transform', `rotate(${degrees * 1.65})`);
        parts['front-wheel'].setAttribute('transform', `rotate(${degrees * 1.65})`);
        parts.crank.setAttribute('transform', `translate(546 509) rotate(${degrees})`);
        parts.pelican.setAttribute('transform', `translate(0 ${bob})`);
        const near = legPath(502, 343 + bob, 546 + crankX, 498 + crankY);
        parts['near-leg-path'].setAttribute('d', near);
        parts['near-leg-color'].setAttribute('d', near);
        parts['far-leg-path'].setAttribute('d', legPath(488, 341 + bob, 546 - crankX, 498 - crankY));
        parts['near-foot'].setAttribute('transform', `translate(${546 + crankX} ${497 + crankY})`);
        parts['far-foot'].setAttribute('transform', `translate(${546 - crankX} ${497 - crankY})`);
        parts.clouds.setAttribute('transform', `translate(${-Math.sin(elapsed * .065) * 34} 0)`);
        parts.birds.setAttribute('transform', `translate(${-Math.sin(elapsed * .1) * 13} ${Math.sin(elapsed * .45) * 3})`);
        parts['water-lines'].setAttribute('transform', `translate(${Math.sin(elapsed * .2) * 12} 0)`);
        parts['road-movement'].setAttribute('transform', `translate(${-((phase * 1.65 * 94) % 600)} 0)`);
        parts['distant-plants'].setAttribute('transform', `translate(${-((elapsed * 22) % 1170)} 0)`);
        // Gentle scarf flutter and an occasional blink share the same paused clock.
        parts['scarf-tail'].setAttribute('transform', `rotate(${Math.sin(phase * 1.7) * 2.2} 593 225)`);
        const blink = elapsed % 5.3;
        parts.eye.setAttribute('transform', blink > 5.12 ? 'translate(0 120.8) scale(1 .2)' : '');
      }

      function tick(now) {
        frameId = null;
        const delta = previousTime === null ? 0 : Math.min((now - previousTime) / 1000, .06);
        previousTime = now;
        if (playing) { elapsed += delta * speed; draw(); }
        if (bellStart !== null) {
          const progress = (now - bellStart) / 1100;
          parts['bell-rings'].setAttribute('opacity', String(Math.max(0, 1 - progress)));
          if (progress >= 1) { bellStart = null; $('bell-announcement').textContent = ''; }
        }
        if ((playing || bellStart !== null) && !document.hidden) frameId = requestAnimationFrame(tick);
        else previousTime = null;
      }

      function schedule() {
        if (frameId === null && !document.hidden && (playing || bellStart !== null)) frameId = requestAnimationFrame(tick);
      }

      function updatePlayback() {
        playButton.setAttribute('aria-label', playing ? '暂停骑行' : '继续骑行');
        playButton.setAttribute('aria-pressed', String(playing));
        $('pause-icon').hidden = !playing;
        $('play-icon').hidden = playing;
        // SVG elements use the hidden attribute explicitly for consistent browsers.
        $('pause-icon').toggleAttribute('hidden', !playing);
        $('play-icon').toggleAttribute('hidden', playing);
        $('status').textContent = playing ? '正在悠闲兜风' : '停一停，看看海';
        $('status-caption').textContent = playing ? 'ENJOY THE LITTLE THINGS' : 'TAKE A LITTLE BREATHER';
        $('status-dot').style.background = playing ? '#7f9e7c' : '#c49c70';
        previousTime = null;
        schedule();
      }

      playButton.addEventListener('click', () => { playing = !playing; updatePlayback(); });
      speedButtons.forEach(button => button.addEventListener('click', () => {
        speed = Number(button.dataset.speed);
        speedButtons.forEach(option => option.setAttribute('aria-pressed', String(option === button)));
      }));

      $('bell-button').addEventListener('click', async () => {
        bellStart = performance.now();
        $('bell-announcement').textContent = '叮铃！鹈鹕先生向你问好。';
        parts['bell-rings'].setAttribute('opacity', '1');
        schedule();
        // The bell is synthesized locally and only sounds after a user gesture.
        try {
          const AudioContext = window.AudioContext || window.webkitAudioContext;
          if (!AudioContext) return;
          audioContext ??= new AudioContext();
          await audioContext.resume();
          const start = audioContext.currentTime;
          [0, .15].forEach(delay => {
            [1174.66, 2349.32, 3524].forEach((frequency, i) => {
              const oscillator = audioContext.createOscillator();
              const gain = audioContext.createGain();
              oscillator.type = 'sine';
              oscillator.frequency.value = frequency;
              gain.gain.setValueAtTime(0, start + delay);
              gain.gain.linearRampToValueAtTime(.085 / (i + 1), start + delay + .004);
              gain.gain.exponentialRampToValueAtTime(.0001, start + delay + .8);
              oscillator.connect(gain);
              gain.connect(audioContext.destination);
              oscillator.start(start + delay);
              oscillator.stop(start + delay + .85);
              oscillator.onended = () => { oscillator.disconnect(); gain.disconnect(); };
            });
          });
        } catch { /* Visual feedback remains available when audio is unavailable. */ }
      });

      document.addEventListener('visibilitychange', () => {
        previousTime = null;
        if (document.hidden && frameId !== null) { cancelAnimationFrame(frameId); frameId = null; }
        else schedule();
      });
      motionPreference.addEventListener('change', event => {
        if (event.matches) { playing = false; updatePlayback(); }
      });
      draw();
      updatePlayback();
    })();
  </script>
