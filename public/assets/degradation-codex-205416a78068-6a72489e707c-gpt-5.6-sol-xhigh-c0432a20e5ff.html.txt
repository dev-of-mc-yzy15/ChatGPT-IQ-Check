  <script>
    (() => {
      const card = document.querySelector('#animationCard');
      const button = document.querySelector('#toggleMotion');
      const icon = button.querySelector('.icon');
      const label = button.querySelector('.label');
      const rearLeg = document.querySelector('#rearLeg');
      const rearFoot = document.querySelector('#rearFoot');
      const frontLeg = document.querySelector('#frontLeg');
      const frontAccent = document.querySelector('#frontKneeAccent');
      const frontFoot = document.querySelector('#frontFoot');
      const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)');

      let paused = reduceMotion.matches;
      let startedAt = performance.now();
      let pausedAt = paused ? performance.now() : 0;
      let pausedTotal = 0;

      const setButton = () => {
        card.classList.toggle('paused', paused);
        button.setAttribute('aria-pressed', String(paused));
        button.setAttribute('aria-label', paused ? '继续动画' : '暂停动画');
        icon.textContent = paused ? '▶' : 'Ⅱ';
        label.textContent = paused ? '继续' : '暂停';
      };

      function legPath(hipX, hipY, angle, isFront) {
        const crankX = 555;
        const crankY = 540;
        const radius = 39;
        const footX = crankX + Math.cos(angle) * radius;
        const footY = crankY + Math.sin(angle) * radius;
        const midX = (hipX + footX) / 2;
        const midY = (hipY + footY) / 2;
        const dx = footX - hipX;
        const dy = footY - hipY;
        const length = Math.hypot(dx, dy);
        const bend = isFront ? 37 : -30;
        const kneeX = midX + (-dy / length) * bend;
        const kneeY = midY + (dx / length) * bend;
        return { d: `M${hipX} ${hipY} Q${kneeX.toFixed(1)} ${kneeY.toFixed(1)} ${footX.toFixed(1)} ${footY.toFixed(1)}`, footX, footY };
      }

      function drawLegs(now) {
        const elapsed = paused ? pausedAt - startedAt - pausedTotal : now - startedAt - pausedTotal;
        const angle = (elapsed / 820) * Math.PI * 2;
        const front = legPath(522, 353, angle, true);
        const rear = legPath(492, 346, angle + Math.PI, false);

        frontLeg.setAttribute('d', front.d);
        frontAccent.setAttribute('d', front.d);
        rearLeg.setAttribute('d', rear.d);
        frontFoot.setAttribute('d', `M${(front.footX - 4).toFixed(1)} ${front.footY.toFixed(1)}l40 ${(-Math.sin(angle) * 5 + 5).toFixed(1)}`);
        rearFoot.setAttribute('d', `M${(rear.footX - 4).toFixed(1)} ${rear.footY.toFixed(1)}l38 ${(-Math.sin(angle + Math.PI) * 5 + 5).toFixed(1)}`);

        requestAnimationFrame(drawLegs);
      }

      button.addEventListener('click', () => {
        if (paused) {
          pausedTotal += performance.now() - pausedAt;
          paused = false;
          card.classList.add('motion-ok');
        } else {
          pausedAt = performance.now();
          paused = true;
        }
        setButton();
      });

      setButton();
      requestAnimationFrame(drawLegs);
    })();
  </script>
