        <span class="live-dot" id="live-dot" aria-hidden="true"></span>
        <div><strong id="ride-status" aria-live="polite">正在收集沿途的好心情</strong><small>A LITTLE RIDE, A LITTLE JOY.</small></div>
      </div>
      <div class="playback">
        <div class="speeds" role="group" aria-label="骑行速度">
          <button class="speed" type="button" data-speed="0.55" aria-pressed="false" aria-label="慢悠悠，0.55 倍速度">慢悠悠</button>
          <button class="speed" type="button" data-speed="1" aria-pressed="true" aria-label="刚刚好，正常速度">刚刚好</button>
          <button class="speed" type="button" data-speed="1.6" aria-pressed="false" aria-label="追着风，1.6 倍速度">追着风</button>
        </div>
        <button id="toggle-play" class="pause" type="button" aria-pressed="false" aria-label="暂停动画">
          <svg viewBox="0 0 12 12" aria-hidden="true"><path id="play-icon" d="M2 1h3v10H2zM7 1h3v10H7z"/></svg>
          <span id="play-label">歇一会儿</span>
        </button>

    <footer class="footer">
      <span>生活偶尔需要一点无所事事。</span>
      <span>MADE OF SVG &amp; <svg viewBox="0 0 16 16" fill="none" stroke="currentColor" stroke-width="1.2" aria-hidden="true"><path d="M8 13S1 9 1 5a3.5 3.5 0 0 1 7-1 3.5 3.5 0 0 1 7 1c0 4-7 8-7 8Z"/></svg> DAYDREAMS</span>
    </footer>

  <script>
    (() => {
      'use strict';
      const byId = id => document.getElementById(id);
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      let paused = reducedMotion.matches;
      let speed = 1;
      let elapsed = 0;
      let previous = null;
      let frameId = null;
      const cadence = 2.5;
      const wheelRatio = 1.7;
      const wheelRadius = 105;
      const crank = { x: 696, y: 488, r: 35 };
      const refs = Object.fromEntries([
        'rear-wheel', 'front-wheel', 'crank', 'pelican', 'wing', 'wing-feathers',
        'near-leg', 'far-leg', 'leg-highlight', 'near-foot', 'far-foot', 'eye',
        'scarf-tail', 'bike-shadow', 'sky-birds',
        'sailboat', 'wind-one', 'wind-two', 'wind-three'
      ].map(id => [id, byId(id)]));
      const scenery = Array.from(document.querySelectorAll('[data-drift]')).map(el => ({
        el, start: Number(el.dataset.start), y: Number(el.dataset.y),
        ratio: el.dataset.drift === 'fast' ? 1 : 0.32
      }));
      const road = Array.from(document.querySelectorAll('[data-road]'));
      const clouds = Array.from(document.querySelectorAll('[data-cloud-start]')).map(el => ({
        el, start: Number(el.dataset.cloudStart), speed: Number(el.dataset.cloudSpeed)
      }));
      const speedButtons = Array.from(document.querySelectorAll('[data-speed]'));
      const wrap = (value, min, max) => ((value - min) % (max - min) + (max - min)) % (max - min) + min;

      // Two-bone inverse kinematics keeps the knee bent and the foot on its pedal.
      function leg(phase, bob, near) {
        const angle = elapsed * cadence + phase;
        const foot = { x: crank.x + crank.r * Math.cos(angle), y: crank.y + crank.r * Math.sin(angle) - 15 };
        const hip = { x: near ? 677 : 668, y: (near ? 328 : 330) + bob };
        const upper = 104;
        const lower = 109;
        const dx = foot.x - hip.x;
        const dy = foot.y - hip.y;
        const distance = Math.hypot(dx, dy);
        const a = (upper * upper - lower * lower + distance * distance) / (2 * distance);
        const h = Math.sqrt(Math.max(0, upper * upper - a * a));
        const knee = { x: hip.x + a * dx / distance + h * dy / distance, y: hip.y + a * dy / distance - h * dx / distance };
        const path = `M${hip.x} ${hip.y}L${knee.x.toFixed(2)} ${knee.y.toFixed(2)}L${foot.x.toFixed(2)} ${foot.y.toFixed(2)}`;
        refs[near ? 'near-leg' : 'far-leg'].setAttribute('d', path);
        if (near) refs['leg-highlight'].setAttribute('d', path);
        refs[near ? 'near-foot' : 'far-foot'].setAttribute('transform', `translate(${foot.x.toFixed(2)} ${foot.y.toFixed(2)})`);
      }

      function render() {
        const angle = elapsed * cadence;
        const degrees = angle * 180 / Math.PI;
        const bob = Math.sin(angle * 2) * 2.2;
        const distance = angle * wheelRatio * wheelRadius;
        refs['rear-wheel'].setAttribute('transform', `rotate(${degrees * wheelRatio})`);
        refs['front-wheel'].setAttribute('transform', `rotate(${degrees * wheelRatio})`);
        refs.crank.setAttribute('transform', `rotate(${degrees} 696 488)`);
        refs.pelican.setAttribute('transform', `translate(0 ${bob})`);
        leg(0, bob, true);
        leg(Math.PI, bob, false);

        // The wing's shoulder moves with the body; the feathers hold the handlebar.
        refs.wing.setAttribute('d', `M657 ${267 + bob}c24-20 52-12 77 9l51 ${40 - bob}q29 14 65-13 10-2 7 6c-22 31-51 37-77 23l-66 ${-30 + bob}c-20 20-55 14-71 2-8-7-5-23 14-37Z`);
        refs['wing-feathers'].setAttribute('transform', `translate(0 ${bob})`);
        refs['scarf-tail'].setAttribute('d', `M764 218C734 ${204 + Math.sin(angle * 1.7) * 3} 722 ${222 + Math.sin(angle * 1.7 + 1) * 4} 688 ${208 + Math.sin(angle * 1.7 + 2) * 5}l7 14-12 8c35 13 51-9 79 6Z`);
        const blink = elapsed % 5.4;
        const eyeScale = blink > 4.98 && blink < 5.16 ? Math.max(.08, Math.abs(blink - 5.07) / .09) : 1;
        refs.eye.setAttribute('transform', `translate(795 151) scale(1 ${eyeScale})`);
        refs['bike-shadow'].setAttribute('rx', 248 + bob * 1.4);

        scenery.forEach(({ el, start, y, ratio }) => {
          el.setAttribute('transform', `translate(${wrap(start - distance * ratio, -100, 1540).toFixed(2)} ${y})`);
        });
        road.forEach((el, i) => el.setAttribute('transform', `translate(${wrap(i * 265 + 150 - distance, -80, 1510).toFixed(2)} 0)`));
        clouds.forEach(cloud => cloud.el.setAttribute('transform', `translate(${wrap(cloud.start - elapsed * cloud.speed, -200, 1600) - cloud.start} 0)`));
        refs['sky-birds'].setAttribute('transform', `translate(${-Math.sin(elapsed * .14) * 25} ${Math.sin(elapsed * 1.2) * 2})`);
        refs.sailboat.setAttribute('transform', `translate(${1169 + Math.sin(elapsed * .12) * 15} ${338 + Math.sin(elapsed * .8) * 1.5})`);
        ['wind-one', 'wind-two', 'wind-three'].forEach((id, index) => {
          const phase = elapsed * 2 + index * 1.4;
          refs[id].setAttribute('transform', `translate(${-Math.sin(phase) * 12} 0)`);
          refs[id].setAttribute('opacity', .35 + (Math.sin(phase) + 1) * .3);
        });
      }

      function frame(now) {
        frameId = null;
        if (paused || document.hidden) { previous = null; return; }
        if (previous !== null) elapsed += Math.min((now - previous) / 1000, .06) * speed;
        previous = now;
        render();
        frameId = requestAnimationFrame(frame);
      }

      function schedule() {
        previous = null;
        if (!paused && !document.hidden && frameId === null) frameId = requestAnimationFrame(frame);
      }

      function updatePlayback() {
        byId('toggle-play').setAttribute('aria-pressed', String(paused));
        byId('toggle-play').setAttribute('aria-label', paused ? '播放动画' : '暂停动画');
        byId('play-label').textContent = paused ? '继续出发' : '歇一会儿';
        byId('play-icon').setAttribute('d', paused ? 'M3 1l8 5-8 5z' : 'M2 1h3v10H2zM7 1h3v10H7z');
        byId('ride-status').textContent = paused ? '停下来，也是一种好风景' : '正在收集沿途的好心情';
        byId('live-dot').style.background = paused ? '#b6b9a7' : '#80997b';
        if (paused && frameId !== null) { cancelAnimationFrame(frameId); frameId = null; }
        schedule();
      }

      byId('toggle-play').addEventListener('click', () => { paused = !paused; updatePlayback(); });
      speedButtons.forEach(button => button.addEventListener('click', () => {
        speed = Number(button.dataset.speed);
        speedButtons.forEach(item => item.setAttribute('aria-pressed', String(item === button)));
      }));
      document.addEventListener('visibilitychange', schedule);
      reducedMotion.addEventListener('change', event => { paused = event.matches; updatePlayback(); });
      render();
      updatePlayback();
    })();
  </script>
</body>
