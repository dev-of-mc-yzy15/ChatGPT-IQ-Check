    (() => {

      const parts = Object.fromEntries([

      // Solve a two-segment leg so each webbed foot stays attached to its pedal.
      function drawLeg(side, angle, bob) {
        const hip = { x: side === 'near' ? 553 : 537, y: 322 + bob };
        const pedal = { x: 590 + Math.cos(angle) * 34, y: 447 + Math.sin(angle) * 34 };
        const ankle = { x: pedal.x - 2, y: pedal.y - 7 };
        const dx = ankle.x - hip.x, dy = ankle.y - hip.y;
        const distance = Math.hypot(dx, dy);
        const upper = 83, lower = 88;
        const projection = (upper * upper - lower * lower + distance * distance) / (2 * distance);
        const height = Math.sqrt(Math.max(0, upper * upper - projection * projection));
        const knee = { x: hip.x + dx / distance * projection + dy / distance * height,
                       y: hip.y + dy / distance * projection - dx / distance * height };
        const path = `M${hip.x.toFixed(2)} ${hip.y.toFixed(2)} L${knee.x.toFixed(2)} ${knee.y.toFixed(2)} L${ankle.x.toFixed(2)} ${ankle.y.toFixed(2)}`;
        parts[`${side}-leg-outline`].setAttribute('d', path);
        parts[`${side}-leg-fill`].setAttribute('d', path);
        setTransform(`${side}-foot`, `translate(${ankle.x.toFixed(2)} ${ankle.y.toFixed(2)})`);
        parts[`${side}-pedal`].setAttribute('d', `M${pedal.x - 15} ${pedal.y + 2}h30`);
        return pedal;
      }

      function render(t) {
        const angle = t * 2.35 + .65;
        const bob = Math.sin(angle * 2) * 1.7;
        const wheelDegrees = (t * 116 / 68 * 180 / Math.PI) % 360;
        setTransform('rear-spokes', `rotate(${wheelDegrees})`);
        setTransform('front-spokes', `rotate(${wheelDegrees})`);
        const far = drawLeg('far', angle + Math.PI, bob);
        const near = drawLeg('near', angle, bob);
        parts.crank.setAttribute('d', `M${far.x} ${far.y}L590 447L${near.x} ${near.y}`);
        setTransform('pelican', `translate(0 ${bob})`);
        const blinkPhase = t % 6.7;
        const eyeScale = blinkPhase > 5.9 && blinkPhase < 6.08 ? .12 : 1;
        setTransform('pelican-eye', `translate(646 184) scale(1 ${eyeScale}) translate(-646 -184)`);
        const flutter = Math.sin(t * 5.5) * 4;
        parts['scarf-tail'].setAttribute('d', `M601 238q-31 ${-15 + flutter}-58 ${-6 + flutter}l11 12-13 10q33 ${-9 - flutter}63 ${-2 - flutter}`);

        });

        setTransform('back-plant-two', `translate(${wrap(911 - t * 70, -70, 1350)} 431)`);

      function syncPlayButton() {
        document.body.classList.toggle('is-paused', !state.playing);
        byId('play-button').setAttribute('aria-label', state.playing ? '暂停动画' : '播放动画');

      function tick(now) {

        const AudioContext = window.AudioContext || window.webkitAudioContext;

            [0, .18].forEach((delay, index) => {
              const oscillator = audioContext.createOscillator();
              const gain = audioContext.createGain();
              oscillator.type = 'sine';
              oscillator.frequency.setValueAtTime(index ? 1760 : 2349, start + delay);
              gain.gain.setValueAtTime(0, start + delay);
              gain.gain.linearRampToValueAtTime(.12, start + delay + .007);
              gain.gain.exponentialRampToValueAtTime(.001, start + delay + .6);
              oscillator.connect(gain);
              gain.connect(audioContext.destination);
              oscillator.start(start + delay);
              oscillator.stop(start + delay + .65);
              oscillator.onended = () => { oscillator.disconnect(); gain.disconnect(); };

        } catch (_) { /* The visual bell remains available if audio is unsupported. */ }

    })();
