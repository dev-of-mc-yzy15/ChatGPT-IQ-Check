    <script>
      (() => {
        const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
        const rearSpokes = document.getElementById('rear-spokes');
        const frontSpokes = document.getElementById('front-spokes');
        const crank = document.getElementById('crank');
        const pelican = document.getElementById('pelican');
        const legs = [
          {
            path: document.getElementById('near-leg-path'),
            knee: document.getElementById('near-knee'),
            foot: document.getElementById('near-foot'),
            hip: { x: 535, y: 319 },
            offset: 0
          },
          {
            path: document.getElementById('far-leg-path'),
            knee: document.getElementById('far-knee'),
            foot: document.getElementById('far-foot'),
            hip: { x: 516, y: 318 },
            offset: Math.PI
          }
        ];

        const hub = { x: 580, y: 510 };
        const crankRadius = 36;
        const upperLeg = 130;
        const lowerLeg = 115;
        const initialPhase = 0;
        let animationStart = null;
        let frameId = null;

        function setLeg(leg, angle, bounce) {
          const hip = { x: leg.hip.x, y: leg.hip.y + bounce };
          const pedal = {
            x: hub.x + crankRadius * Math.cos(angle),
            y: hub.y + crankRadius * Math.sin(angle)
          };
          const dx = pedal.x - hip.x;
          const dy = pedal.y - hip.y;
          const distance = Math.hypot(dx, dy);
          const along = (upperLeg ** 2 - lowerLeg ** 2 + distance ** 2) / (2 * distance);
          const outward = Math.sqrt(Math.max(0, upperLeg ** 2 - along ** 2));
          const knee = {
            x: hip.x + (along * dx + outward * dy) / distance,
            y: hip.y + (along * dy - outward * dx) / distance
          };

          leg.path.setAttribute('d', `M${hip.x.toFixed(1)} ${hip.y.toFixed(1)}L${knee.x.toFixed(1)} ${knee.y.toFixed(1)}L${pedal.x.toFixed(1)} ${pedal.y.toFixed(1)}`);
          leg.knee.setAttribute('cx', knee.x.toFixed(1));
          leg.knee.setAttribute('cy', knee.y.toFixed(1));
          leg.foot.setAttribute('transform', `translate(${pedal.x.toFixed(1)} ${pedal.y.toFixed(1)}) rotate(-8)`);
        }

        function draw(seconds) {
          const phase = initialPhase + seconds * Math.PI * 1.7;
          const degrees = phase * 180 / Math.PI;
          const bounce = Math.sin(phase * 2) * 2.5;

          pelican.setAttribute('transform', `translate(0 ${bounce.toFixed(2)})`);
          crank.setAttribute('transform', `rotate(${degrees.toFixed(2)} ${hub.x} ${hub.y})`);
          rearSpokes.setAttribute('transform', `rotate(${(degrees * 2.15).toFixed(2)})`);
          frontSpokes.setAttribute('transform', `rotate(${(degrees * 2.15).toFixed(2)})`);
          legs.forEach(leg => setLeg(leg, phase + leg.offset, bounce));
        }

        function tick(now) {
          if (reducedMotion.matches) return;
          if (animationStart === null) animationStart = now;
          draw((now - animationStart) / 1000);
          frameId = requestAnimationFrame(tick);
        }

        function syncMotionPreference() {
          if (frameId !== null) cancelAnimationFrame(frameId);
          frameId = null;
          animationStart = null;
          draw(0);
          if (!reducedMotion.matches) frameId = requestAnimationFrame(tick);
        }

        reducedMotion.addEventListener('change', syncMotionPreference);
        syncMotionPreference();
      })();
    </script>
