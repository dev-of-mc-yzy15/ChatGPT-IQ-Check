<script>
const scene=document.getElementById('scene'),button=document.getElementById('toggle');
const reduced=matchMedia('(prefers-reduced-motion: reduce)');
let paused=reduced.matches,time=0,last;
const part=id=>document.getElementById(id);
function render(){
 const angle=time/1200*Math.PI*2;
 for(const [side,offset,hipX] of [['near',0,477],['far',Math.PI,462]]){
  const a=angle+offset,x=490+29*Math.cos(a),y=427+29*Math.sin(a);
  // Two connected segments keep the feet on the circular pedal path.
  const hx=hipX,hy=332,dx=x-hx,dy=y-hy,d=Math.hypot(dx,dy),length=67;
  const bend=Math.sqrt(Math.max(0,length*length-d*d/4));
  const kx=(hx+x)/2+dy/d*bend,ky=(hy+y)/2-dx/d*bend;
  part(side+'Leg').setAttribute('d',`M${hx} ${hy}L${kx} ${ky}L${x} ${y}`);
  part(side+'Pedal').setAttribute('d',`M${x-12} ${y+7}H${x+12}`);
  if(side==='near')part('nearFoot').setAttribute('d',`M${x-7} ${y}H${x+16}`);
 }
}
function update(){scene.classList.toggle('paused',paused);button.textContent=paused?'播放动画':'暂停动画';button.setAttribute('aria-pressed',String(paused));}
button.addEventListener('click',()=>{paused=!paused;if(!paused&&reduced.matches){scene.querySelectorAll('*').forEach(el=>el.style.animationPlayState='running');}update();});
reduced.addEventListener('change',e=>{paused=e.matches;scene.querySelectorAll('*').forEach(el=>el.style.animationPlayState='');update();});
function frame(now){if(last!==undefined&&!paused)time+=Math.min(now-last,50);last=now;render();requestAnimationFrame(frame);}
update();requestAnimationFrame(frame);
</script>
