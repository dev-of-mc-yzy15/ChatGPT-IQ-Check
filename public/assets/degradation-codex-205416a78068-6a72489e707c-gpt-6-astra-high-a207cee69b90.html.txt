  <script>
    (() => {
      const $ = id => document.getElementById(id);
      const motion = window.matchMedia('(prefers-reduced-motion: reduce)');
      let playing = !motion.matches, speed = 1, time = 0, previous = null, frame = null;
      const speeds = [1, 1.5, 0.5];
      function draw(t) {
        const angle = t * 2.7;
        const degrees = angle * 180 / Math.PI;
        const bob = Math.sin(angle * 2) * 1.7;
        $('rear-wheel').setAttribute('transform', `rotate(${degrees * 1.35})`);
        $('front-wheel').setAttribute('transform', `rotate(${degrees * 1.35})`);
        $('crank').setAttribute('transform', `rotate(${degrees} 530 426)`);
        $('pelican').setAttribute('transform', `translate(0 ${bob})`);
        $('scarf-tail').setAttribute('transform', `rotate(${Math.sin(angle * 2.2) * 5} 545 193)`);
        for (const [side, offset, hipX] of [['near', 0, 496], ['far', Math.PI, 484]]) {
          const x = 530 + Math.cos(angle + offset) * 25;
          const y = 426 + Math.sin(angle + offset) * 25;
          const hipY = 280 + bob;
          // Two equal-length leg segments keep the foot attached to its pedal.
          const dx = x - hipX, dy = y - hipY, distance = Math.hypot(dx, dy);
          const kneeOffset = Math.sqrt(Math.max(0, 91 * 91 - distance * distance / 4));
          const kx = (hipX + x) / 2 + dy / distance * kneeOffset;
          const ky = (hipY + y) / 2 - dx / distance * kneeOffset;
          $(`${side}-leg-path`).setAttribute('d', `M${hipX} ${hipY} L${kx} ${ky} L${x} ${y}`);
          $(`${side}-foot`).setAttribute('d', `M${x - 7} ${y} Q${x + 6} ${y - 3} ${x + 17} ${y + 2}`);
          $(`${side}-pedal`).setAttribute('d', `M${x - 10} ${y + 9}h29`);
        }
        $('clouds').setAttribute('transform', `translate(${-((t * 5) % 1160)} 0)`);
        // Individual scenery elements wrap beyond the visible edges.
        [...$('clouds').children].forEach((cloud, i) => {
          const initial = [180, 678, 875][i];
          const x = ((initial - t * 5 + 1160 * 100) % 1160) - 30;
          cloud.setAttribute('transform', `translate(${x - initial + ((t * 5) % 1160)} 0)`);
        });
        [...$('plants').children].forEach((plant, i) => {
          const x = (([190, 913][i] - t * 90) % 1160 + 1160) % 1160 - 30;
          plant.setAttribute('transform', `translate(${x} 524) scale(${i ? .65 : 1})`);
          plant.setAttribute('x', '0'); plant.setAttribute('y', '0');
        });
        $('road').setAttribute('transform', `translate(${-((t * 40) % 120)} 0)`);
        $('ripples').setAttribute('transform', `translate(${Math.sin(t * .6) * 10} 0)`);
      }
      function tick(now) {
        if (!playing) return;
        if (previous !== null) time += Math.min((now - previous) / 1000, .05) * speed;
        previous = now; draw(time); frame = requestAnimationFrame(tick);
      }
      function updatePlayback() {
        $('play').setAttribute('aria-pressed', String(playing));
        $('play-label').textContent = playing ? '歇一会儿' : '继续骑行';
        $('play-icon').setAttribute('d', playing ? 'M2 1h3v10H2zM7 1h3v10H7z' : 'M3 1l8 5-8 5Z');
        cancelAnimationFrame(frame); previous = null;
        if (playing) frame = requestAnimationFrame(tick);
      }
      $('play').addEventListener('click', () => { playing = !playing; updatePlayback(); });
      $('speed').addEventListener('click', () => {
        speed = speeds[(speeds.indexOf(speed) + 1) % speeds.length];
        $('speed-label').textContent = `${speed}×`;
        $('speed').setAttribute('aria-label', `切换骑行速度，当前为 ${speed} 倍速`);
      });
      motion.addEventListener('change', event => { playing = !event.matches; updatePlayback(); });
      document.addEventListener('visibilitychange', () => { previous = null; });
      draw(0); updatePlayback();
    })();
  </script>
